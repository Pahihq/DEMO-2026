```bash

    ticket_lifetime = 24h

 AU-TEAM.IRPO = {
     kdc = br-srv.au-team.irpo
     admin_server = br-srv.au-team.irpo
 }

[domain_realm]
 .au-team.irpo = AU-TEAM.IRPO
```

2. Добавляем пользователей и группу
```bash
samba-tool group add left
samba-tool user add left1 P@ssw0rd
samba-tool user add left2 P@ssw0rd
samba-tool user add left3 P@ssw0rd
samba-tool user add left4 P@ssw0rd
samba-tool user add left5 P@ssw0rd

samba-tool group addmembers left left1,left2,left3,left4,left5
```
