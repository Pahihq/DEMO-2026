# На SRV-R

```bash
dnf install ansible
ssh-keygen
ssh-copy-id user@192.168.100.2 # SRV-L
ssh-copy-id user@192.168.200.2 #CLI-L
# Дописываем в начало файла nano /etc/ansible/hosts
[left]
192.168.100.2  ansible_user=sshuser
192.168.200.2  ansible_user=user

# Дописываем в файл nano /etc/ansible/ansible.cfg
[defaults]
host_key_checking = false
interpreter_python=auto_silent  

# Копируем get_hostname_address.yml c cd-rom в директорию etc/ansible
cp /mnt/playbook/get_hostname_address.yml /etc/ansible
# Меняем права на файл
chmod 766 /etc/ansible/get_hostname_address.yml

# Открываем на редактирование /etc/ansible/get_hostname_address.yml
 nano /etc/ansible/get_hostname_address.yml
```
```yml
- name: получение данных с хоста
  hosts: left
  tasks:
    - name: получение данных с хоста
      delegate_to: localhost
      copy:
        dest: "/etc/ansible/PC-INFO/{{ ansible_hostname }}.yml"
        content: |
          Hostname: {{ ansible_hostname }}
          IP_Address: {{ ansible_default_ipv4.address }}
```
```bash
# Создаем каталог PC-INFO
mkdir /etc/ansible/PC-INFO
cd /etc/ansible
ansible-playbook get_hostname_address.yml
```
