# На SRV-R

Импорт пользователей в домен
```bash
# Монтируем Additional.iso
mount /dev/sr0 /mnt/
```

Пишем скрипт
```bash
#!/bin/bash

FILE="/mnt/Users.csv"

while IFS=',' read -r firstname lastname password role phone ou zip country city street
do
    password=${password//[[:space:]]/}
    street=${street//[^[:print:]]/}

    samba-tool user add "${firstname}.${lastname}" "$password" \
        --given-name="$firstname" \
        --surname="$lastname" \
        --job-title="$role" \
        --telephone-number="$phone" \
        --department="$ou" \
        --description="$zip,$country,$city,$street"
done < <(tail -n +2 "$FILE")
```
